/* BLOOD CARD */

function BloodCard({
  group,
  units,
  status,
  color,
  width,
}) {
  const bar =
    color === "green"
      ? "bg-[#238356]"
      : color === "yellow"
      ? "bg-[#c3840c]"
      : "bg-[#c5163f]";

  const text =
    color === "green"
      ? "text-[#18804e]"
      : color === "yellow"
      ? "text-[#bd7a08]"
      : "text-[#c5163f]";

  return (
    <div className="rounded-[19px] border border-[#e6dada] bg-white p-6">

      <div className="text-[22px] font-extrabold">
        {group}
      </div>

      <div className="mt-2 text-[#8b7a7d]">

    donor: {
      title: "Donor dashboard.",
      text: "Manage your donor profile, availability and donation history.",
      button: "Register as a donor",
    },
  };

  const current = data[page];

  const [group, setGroup] = useState("O+");
  const [units, setUnits] = useState("2");
  const [success, setSuccess] = useState("");

  const submit = (e) => {
    e.preventDefault();

    if (page === "find") {
      setSuccess(
        `Searching donors for ${group} near Delhi NCR...`
      );
    }

    if (page === "request") {
      setSuccess(
        `Request created for ${units} unit(s) of ${group}.`
      );
    }

    if (page === "donor") {
      setSuccess(
        "Your donor registration is ready."
      );
    }
  };

  return (
    <main className="mx-auto min-h-screen max-w-[1200px] px-6 py-16 lg:py-24">

      <div className="rounded-[32px] border border-[#e5dada] bg-white p-7 shadow-sm sm:p-12">

        <div className="inline-flex rounded-full bg-[#fbecef] px-4 py-2 font-semibold text-[#c5163f]">
          BloodLink
        </div>

        <h1 className="mt-6 text-5xl font-extrabold sm:text-6xl">
          {current.title}
        </h1>

        <p className="mt-5 max-w-3xl text-lg leading-8 text-[#66585b]">
          {current.text}
        </p>

        <form
          onSubmit={submit}
          className="mt-10 grid max-w-3xl gap-5 sm:grid-cols-2"
        >

          <label className="font-semibold">
            Blood group

            <select
              value={group}
              onChange={(e) => setGroup(e.target.value)}
              className="mt-2 w-full rounded-2xl border p-4 outline-none"
            >

              {bloodGroups.map((blood) => (
                <option key={blood.group}>
                  {blood.group}
                </option>
              ))}

            </select>

          </label>

          {page !== "donor" && (
            <label className="font-semibold">
              Units needed

              <input
                type="number"
                min="1"
                value={units}
                onChange={(e) => setUnits(e.target.value)}
                className="mt-2 w-full rounded-2xl border p-4 outline-none"
              />

            </label>
          )}

          <label className="font-semibold sm:col-span-2">
            Location

            <input
              defaultValue="Delhi NCR"
              className="mt-2 w-full rounded-2xl border p-4 outline-none"
            />

          </label>

          <button
            className="rounded-full bg-[#c5163f] px-7 py-4 font-bold text-white sm:w-fit"
          >
            {current.button}
          </button>

        </form>

        {success && (
          <div className="mt-6 rounded-2xl bg-[#eef8f2] p-4 font-semibold text-[#19764a]">
            {success}
          </div>
        )}

        <button
          onClick={() => navigate("home")}
          className="mt-10 rounded-full border px-6 py-3 font-semibold"
        >
          ← Back to Home
        </button>

      </div>

    </main>
  );
}
